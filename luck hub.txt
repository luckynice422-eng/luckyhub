-- ยิง remote ได้ทั้ง RemoteEvent / RemoteFunction (ไม่ error ถ้าเป็นชนิดอื่น)
local function fire(remote, ...)
    if not remote then return false end
    if remote:IsA("RemoteEvent") then
        remote:FireServer(...)
        return true
    elseif remote:IsA("RemoteFunction") then
        local args = table.pack(...)
        task.spawn(function()
            pcall(remote.InvokeServer, remote, table.unpack(args, 1, args.n))
        end)
        return true
    end
    return false
end

-- แคช Level label (ค้นหา recursive เฉพาะตอนยังไม่เจอ และไม่ถี่เกิน 3 วินาที)
local levelLabel, lastLabelSearch = nil, 0
local function updateLevel()
    if not (levelLabel and levelLabel.Parent) then
        if os.clock() - lastLabelSearch < 3 then return end
        lastLabelSearch = os.clock()
        -- ตำแหน่งจริงของ label เลเวล (ยืนยันจากไฟล์ dump): WK_HUD_Safe > Container > Identity Card > Level
        local safe = playerGui:FindFirstChild("WK_HUD_Safe")
        local container = safe and safe:FindFirstChild("Container")
        local card = container and container:FindFirstChild("Identity Card")
        local found = card and card:FindFirstChild("Level")
        if not (found and found:IsA("TextLabel")) then
            found = nil
            -- สำรอง: ค้นทั้ง PlayerGui เผื่อเกมย้าย label
            for _, d in ipairs(playerGui:GetDescendants()) do
                if d.Name == "Level" and d:IsA("TextLabel") then found = d break end
            end
        end
        if found then levelLabel = found else return end
    end
    -- ตัดเครื่องหมาย , ออกก่อน ("Lv. 1,234" -> 1234)
    local digits = (levelLabel.Text:gsub(",", "")):match("%d+")
    local parsed = tonumber(digits)
    if parsed then cachedLevel = parsed end
end

-- [[ ระบบลูป: token ป้องกันลูปซ้อน + pcall ครอบทุกรอบ ]]
local LoopTokens = {}

local function startLoop(name, isEnabled, step)
    local token = {}
    LoopTokens[name] = token
    local function isAlive()
        return Runtime.alive and LoopTokens[name] == token and isEnabled()
    end
    task.spawn(function()
        while isAlive() do
            local ok, err = pcall(step, isAlive)
            if not ok then
                warn("[LuckyHub] " .. name .. ": " .. tostring(err))
                task.wait(1)
            end
        end
    end)
end

local function stopLoop(name)
    LoopTokens[name] = nil
end

local function matchesName(v, name)

end

local function matchesIsland(v, island)
    local a = v:GetAttribute("WK_IslandName")
    return a == nil or a == island
end

-- หา part สำหรับล็อกตำแหน่ง/เล็ง: HumanoidRootPart -> WK_HitPart (เช่น PumpkinKing) -> PrimaryPart -> part ใดก็ได้
local function getEnemyRoot(enemy)
    local part = enemy:FindFirstChild("HumanoidRootPart")
    if not part then
        local hitName = enemy:GetAttribute("WK_HitPart")
        if type(hitName) == "string" then part = enemy:FindFirstChild(hitName, true) end
    end
    part = part or enemy.PrimaryPart or enemy:FindFirstChildWhichIsA("BasePart", true)
    if part and part:IsA("BasePart") then return part end
    return nil
end

-- มอนที่สู้ไม่จบ (เช่น ตีไม่เข้า) จะถูกข้ามชั่วคราว ไม่งั้น "ใกล้สุด" จะเลือกตัวเดิมซ้ำตลอด
local Blacklist = setmetatable({}, { __mode = "k" })

-- หามอนที่ตรงเงื่อนไขและ "ใกล้ที่สุดจากตัวเรา" (ไม่วิ่งข้ามแมพ)
-- opts.allowImmune : รวมมอนที่อมตะ (WK_DamageImmune) ด้วย
-- opts.center/radius : รับเฉพาะมอนที่อยู่ในรัศมีรอบจุด center (ใช้หามอนบริวารรอบบอส)
local function getNearestEnemy(matchFn, opts)
    opts = opts or {}
    local myHRP = getMyHRP()
    if not (EnemiesFolder and myHRP) then return nil end
    local best, bestDist
    for _, v in ipairs(EnemiesFolder:GetChildren()) do
        local skipUntil = Blacklist[v]
        if v:IsA("Model") and not (skipUntil and skipUntil > os.clock()) and matchFn(v)
            and v:GetAttribute("WK_PooledInactive") ~= true -- ตัวที่อยู่ใน pool ยังไม่เกิดจริง
            and (opts.allowImmune or v:GetAttribute("WK_DamageImmune") ~= true) then
            local hrp = getEnemyRoot(v)
            if hrp and getHealth(v) > 0 then
                local inRange = true
                if opts.radius and opts.center then
                    inRange = (hrp.Position - opts.center).Magnitude <= opts.radius
                end
                if inRange then
                    local d = (hrp.Position - myHRP.Position).Magnitude
                    if not bestDist or d < bestDist then
                        best, bestDist = v, d
                    end
                end
            end
        end
    end
    return best
end

-- เลือกเป้าหมายที่ควรตีตอนนี้ รองรับบอสที่อมตะจนกว่าจะเคลียร์มอนรอบตัว (เช่น RadesBoss)
-- คืนค่า: enemy, ชนิด ("target" = ตีตามปกติ | "add" = มอนบริวารรอบบอสอมตะ | "immune" = บอสอมตะและไม่พบบริวาร | "none")
local function pickTarget(matchFn)
    local target = getNearestEnemy(matchFn)
    if target then return target, "target" end

    local boss = getNearestEnemy(matchFn, { allowImmune = true })
    if not boss then return nil, "none" end -- ไม่มีเลย หรือมอนยังไม่โหลดเข้ามาใกล้ (streaming)

    local bossRoot = getEnemyRoot(boss)
    local add = bossRoot and getNearestEnemy(function(v) return v ~= boss end, {
        center = bossRoot.Position,
        radius = Settings.BossAddRadius,
    })
    if add then return add, "add" end
    return nil, "immune"
end

-- [[ สกิล ]]
-- รูปแบบ argument ยืนยันจากล็อก remote spy:
--   สกิลเล็งเป้า : CastSkill(ชื่อ, CFrame, จุดเป้าหมาย)   เช่น DivineRain, LightPrison
--   สกิลไม่เล็ง  : CastSkill(ชื่อ, CFrame)                เช่น LightSwords
local UntargetedSkills = { LightSwords = true }
local skillCache = { raw = nil, list = {} }
local skillState = { index = 1, last = 0 }
local comboState = { index = 1 } -- BasicAttack ตัวที่ 2 ที่เกมส่งจริงสลับ 1,2,1,2 (ยืนยันจากล็อก)

local function getSkillList()
    if skillCache.raw ~= Settings.SkillList then
        skillCache.raw = Settings.SkillList
        local list = {}
        for piece in string.gmatch(Settings.SkillList, "[^,]+") do
            local name = piece:match("^%s*(.-)%s*$")
            if name ~= "" then table.insert(list, name) end
        end
        skillCache.list = list
        skillState.index = 1
    end
    return skillCache.list
end

local function castNextSkill(aimCFrame, targetPos)
    if not CastSkillRemote then return end
    local list = getSkillList()
    if #list == 0 then return end
    if skillState.index > #list then skillState.index = 1 end
    local name = list[skillState.index]
    skillState.index = skillState.index + 1
    if UntargetedSkills[name] then
        fire(CastSkillRemote, name, aimCFrame)
    else
        fire(CastSkillRemote, name, aimCFrame, targetPos)
    end
end

-- [[ เช็กเลือดตัวเอง ]]
local function getMyHpPercent()
    local char = player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum and hum.MaxHealth > 0 then
        return hum.Health / hum.MaxHealth * 100
    end
    return 100
end

-- เลือดต่ำกว่าที่ตั้งไว้ -> ลอยขึ้นสูงหนีมอน รอฟื้น (สูงสุด 60 วินาทีแล้วกลับไปสู้ต่อ)
local function retreatUntilHealed(isAlive)
    local myHRP = getMyHRP()
    if myHRP then myHRP.CFrame = myHRP.CFrame + Vector3.new(0, 80, 0) end
    Status.state = "HP ต่ำ - พักฟื้น"
    local resumeAt = math.min(Settings.MinHpPercent + 25, 95)
    local t0 = os.clock()
    while isAlive() and getMyHpPercent() < resumeAt and os.clock() - t0 < 60 do
        task.wait(0.5)
    end
end

-- ฟังก์ชันโจมตีตัวเดียว ใช้ร่วมกันทั้งสองโหมดฟาร์ม
local function fightEnemy(enemy, isAlive)
    local hrp = getEnemyRoot(enemy)
    if not hrp then return end
    local startTime = os.clock()
    while isAlive() and enemy.Parent and hrp.Parent and getHealth(enemy) > 0 do
        -- เป้าหมายกลายเป็นอมตะระหว่างสู้ (เปลี่ยนเฟส) -> หยุดตี ให้ลูปหลักเลือกเป้าใหม่
        if enemy:GetAttribute("WK_DamageImmune") == true then break end
        local myHRP = getMyHRP()
        if not myHRP then break end -- ตายอยู่ ออกไปให้ลูปหลักตรวจใหม่

        if Settings.MinHpPercent > 0 and getMyHpPercent() < Settings.MinHpPercent then
            retreatUntilHealed(isAlive)
            startTime = os.clock() -- ไม่นับเวลาพักเป็นเวลาต่อสู้
        else
            local myPos, targetPos = getFarmPosition(hrp)
            local aimCFrame = CFrame.lookAt(myPos, targetPos)
            myHRP.CFrame = aimCFrame
            if BasicAttackEvent then
                if player:GetAttribute("WK_ActiveCombatStyle") == "Fists" then
                    -- สไตล์หมัด: เกมส่งแค่ CFrame ตัวเดียว (ยืนยันจากล็อก)
                    BasicAttackEvent:FireServer(aimCFrame)
                else
                    -- สไตล์ Magic: (CFrame, คอมโบ 1/2 สลับกัน, จุดเป้าหมาย, 0) (ยืนยันจากล็อก)
                    comboState.index = (comboState.index == 1) and 2 or 1
                    BasicAttackEvent:FireServer(aimCFrame, comboState.index, targetPos, 0)
                end
            end
            if Settings.AutoSkill and os.clock() - skillState.last >= Settings.SkillInterval then
                skillState.last = os.clock()
                castNextSkill(aimCFrame, targetPos)
            end
            task.wait(0.1)
        end

        if os.clock() - startTime > 15 then
            Blacklist[enemy] = os.clock() + 10
            break
        end
    end
    if not enemy.Parent or getHealth(enemy) <= 0 then
        Status.kills = Status.kills + 1
    end
end

-- ไปยืนข้าง NPC ให้เควสต์ (ใช้เมื่อรับเควสต์จากระยะไกลไม่ติด)
-- ชื่อ NPC อ้างอิงจากโครงสร้างเกม: แต่ละเกาะมี MobNPC (เควสต์ Kill) และ BossNPC (เควสต์ Boss)
local function goToQuestNpc(q)
    local island = workspace:FindFirstChild(q.island)
    local npcName = string.find(q.quest, "_Boss$") and "BossNPC" or "MobNPC"
    local npc = island and island:FindFirstChild(npcName)
    local myHRP = getMyHRP()
    if not (npc and myHRP) then return false end
    local ok, pivot = pcall(function() return npc:GetPivot() end)
    if not ok then return false end
    myHRP.CFrame = pivot * CFrame.new(0, 2, 5)
    task.wait(0.6)
    return true
end

-- [[ เก็บดรอปอัตโนมัติ ]]
-- ดรอปของเกมอยู่ใน Workspace.WK_MaterialDrops / WK_GrimoireDrops / WK_ItemDrops
-- แต่ละชิ้นมี ProximityPrompt (MaterialNode มี attribute WK_MatId บอกชนิดวัตถุดิบ)
local DropFolderNames = { "WK_MaterialDrops", "WK_GrimoireDrops", "WK_ItemDrops" }
local collectTried = setmetatable({}, { __mode = "k" }) -- prompt -> เวลาที่ลองเก็บล่าสุด

local function promptPosition(prompt)
    local p = prompt.Parent
    if not p then return nil end
    if p:IsA("BasePart") then return p.Position end
    if p:IsA("Attachment") then return p.WorldPosition end
    if p:IsA("Model") then return p:GetPivot().Position end
    return nil
end

local function getNearestDropPrompt(range)
    local myHRP = getMyHRP()
    if not myHRP then return nil end
    local best, bestDist = nil, range
    for _, name in ipairs(DropFolderNames) do
        local folder = workspace:FindFirstChild(name)
        if folder then
            for _, d in ipairs(folder:GetDescendants()) do
                if d:IsA("ProximityPrompt") and d.Enabled then
                    local tried = collectTried[d]
                    if not (tried and os.clock() - tried < 5) then
                        local pos = promptPosition(d)
                        if pos then
                            local dist = (pos - myHRP.Position).Magnitude
                            if dist <= bestDist then best, bestDist = d, dist end
                        end
                    end
                end
            end
        end
    end
    return best
end

local function triggerPrompt(prompt)
    if fireproximityprompt then
        fireproximityprompt(prompt)
    else
        prompt:InputHoldBegin()
        task.wait((prompt.HoldDuration or 0) + 0.1)
        prompt:InputHoldEnd()
    end
end

-- เก็บดรอปในระยะ CollectRange (สูงสุด 8 ชิ้นต่อครั้ง เพื่อไม่ให้ค้างอยู่กับการเก็บนานเกิน)
local function collectNearbyDrops(isAlive)
    local count = 0
    while count < 8 and isAlive() do
        local prompt = getNearestDropPrompt(Settings.CollectRange)
        if not prompt then break end
        local pos = promptPosition(prompt)
        local myHRP = getMyHRP()
        if not (pos and myHRP) then break end
        collectTried[prompt] = os.clock() -- ชิ้นที่เก็บไม่หาย จะไม่ถูกลองซ้ำภายใน 5 วินาที
        myHRP.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
        task.wait(0.15)
        pcall(triggerPrompt, prompt)
        task.wait(0.25)
        count = count + 1
    end
end

local function New(class, props, parent)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do inst[k] = v end
    inst.Parent = parent
    return inst
end

local function corner(inst, r)
    return New("UICorner", { CornerRadius = (typeof(r) == "UDim") and r or UDim.new(0, r) }, inst)
end

local function stroke(inst, color, thickness)
    return New("UIStroke", {
        Color = color,
        Thickness = thickness,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, inst)
end

local function isPointer(input)
    return input.UserInputType == Enum.UserInputType.MouseButton1

local function isPointerMove(input)

        or input.UserInputType == Enum.UserInputType.Touch

local function createCardToggle(parent, titleText, descText, default, callback)
    local Card = createCard(parent, 48)

        TextColor3 = UITheme.SubText,
        Font = Enum.Font.Gotham,
        TextSize = 10,
        TextXAlignment = Enum.TextXAlignment.Left,
        BackgroundTransparency = 1,
    }, Card)

    local state = default and true or false

    local Switch = New("TextButton", {
        Text = "",
        Size = UDim2.fromOffset(38, 20),
        Position = UDim2.new(1, -48, 0.5, -10),
        BackgroundColor3 = state and UITheme.Accent or UITheme.ToggleOff,
    }, Card)
    corner(Switch, UDim.new(1, 0))

    local Knob = New("Frame", {
        Size = UDim2.fromOffset(14, 14),
        Position = state and UDim2.new(1, -17, 0.5, -7) or UDim2.new(0, 3, 0.5, -7),
        BackgroundColor3 = Color3.fromRGB(255, 255, 255),
    }, Switch)
    corner(Knob, UDim.new(1, 0))

    local function applyVisual()
        TweenService:Create(Switch, TweenInfo.new(0.2), {
            BackgroundColor3 = state and UITheme.Accent or UITheme.ToggleOff,
        }):Play()
        TweenService:Create(Knob, TweenInfo.new(0.2), {
            Position = state and UDim2.new(1, -17, 0.5, -7) or UDim2.new(0, 3, 0.5, -7),
        }):Play()
    end

    Switch.MouseButton1Click:Connect(function()
        state = not state
        applyVisual()
        callback(state)
        saveConfig()
    end)

    local api = {}
    function api.Set(v)
        state = v and true or false
        applyVisual()
    end

    if state then task.defer(callback, true) end
    return api
end

    local Card = createCard(parent, 46)
    default = math.clamp(default, min, max)

    New("TextLabel", {
        Text = titleText,
        Size = UDim2.new(0.6, 0, 0, 16),
        Position = UDim2.new(0, 12, 0, 5),
        TextColor3 = UITheme.Text,
        Font = Enum.Font.GothamMedium,
        TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Left,
        BackgroundTransparency = 1,
    }, Card)

    local ValLabel = New("TextLabel", {
        Text = tostring(default),
        Size = UDim2.new(0.4, -12, 0, 16),
        Position = UDim2.new(0.6, 0, 0, 5),
        TextColor3 = UITheme.Accent,
        Font = Enum.Font.GothamBold,
        TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Right,
        BackgroundTransparency = 1,
    }, Card)

    local SliderBar = New("Frame", {
        Size = UDim2.new(1, -24, 0, 4),
        Position = UDim2.new(0, 12, 0, 30),
        BackgroundColor3 = UITheme.Sidebar,
    }, Card)
    corner(SliderBar, UDim.new(1, 0))

    local Fill = New("Frame", {
        Size = UDim2.new(math.clamp((default - min) / (max - min), 0, 1), 0, 1, 0),
        BackgroundColor3 = UITheme.Accent,
    }, SliderBar)
    corner(Fill, UDim.new(1, 0))

    local dragging = false
    local function update(input)
        local width = math.max(SliderBar.AbsoluteSize.X, 1)
        local pos = math.clamp((input.Position.X - SliderBar.AbsolutePosition.X) / width, 0, 1)
        Fill.Size = UDim2.new(pos, 0, 1, 0)
        local val = math.floor(min + ((max - min) * pos) + 0.5) -- ปัดเศษ เพื่อให้ถึงค่า max ได้
        ValLabel.Text = tostring(val)
        callback(val)
        saveConfig() -- debounce อยู่ใน saveConfig แล้ว
    end

    SliderBar.InputBegan:Connect(function(input)
        if isPointer(input) then
            dragging = true
            update(input)
        end
    end)
    track(UserInputService.InputEnded:Connect(function(input)
        if isPointer(input) then dragging = false end
    end))
    track(UserInputService.InputChanged:Connect(function(input)
        if dragging and isPointerMove(input) then update(input) end
    end))
end

local farmToggle, selectedToggle, bossToggle

local function autoFarmStep(isAlive)
    updateLevel()

    -- ล็อกเป้าหมายไว้ตราบที่เควสต์ยังค้าง (กันเลเวลข้ามช่วงกลางเควสต์แล้วไปฟาร์มมอนผิดตัว)
    -- คำนวณใหม่จากเลเวลเฉพาะตอนไม่มีเควสต์ หรือยังไม่เคยล็อก (เช่น เปิดฟาร์มตอนเควสต์ค้างอยู่)
    if not farmState.target or not checkQuestActive() then
        farmState.target = getQuestForLevel(cachedLevel)
    end
    local q = farmState.target
    if not q then task.wait(1) return end

    -- ไปเกาะที่ตรงกับเลเวล (ครั้งเดียวต่อเกาะ)
    if TeleportRemote and q.island ~= farmState.lastIsland then
        Status.state = "วาร์ปไป " .. q.island
        fire(TeleportRemote, q.island)
        farmState.lastIsland = q.island
        task.wait(3.5)
        if not isAlive() then return end
    end

    -- รับเควสต์เฉพาะเมื่อไม่มีเควสต์ค้าง (และไม่ยิงถี่กว่า 3 วินาที)
    -- ยืนยันจากล็อก: AcceptQuest("STARTER_Ch1_Kill") ยิงตรงๆ โดยไม่มี InteractNpc นำหน้า
    if AcceptQuest and not checkQuestActive() and os.clock() - farmState.lastAccept > 3 then
        farmState.lastAccept = os.clock()
        Status.state = "รับเควสต์ " .. q.quest
        -- รับไม่ติดติดกันหลายครั้ง (เช่น เซิร์ฟเวอร์เช็กระยะห่างจาก NPC) -> ไปยืนข้าง NPC แล้วรับใหม่
        -- ทำครั้งเดียวต่อเควสต์ กันวาร์ปไปมาถ้าตรวจสถานะเควสต์ผิดพลาด
        if farmState.acceptFails >= 2 and farmState.npcVisited ~= q.quest then
            farmState.npcVisited = q.quest
            goToQuestNpc(q)
        end
        fire(AcceptQuest, q.quest)
        task.wait(1)
        if checkQuestActive() then
            farmState.acceptFails = 0
        else
            farmState.acceptFails = farmState.acceptFails + 1
        end
        if not isAlive() then return end
    end

    -- เก็บดรอปที่เพิ่งตกระหว่างรอบต่อสู้ (ทำระหว่างรอบเท่านั้น ไม่แย่ง CFrame กับตอนสู้)
    if Settings.AutoCollect then
        Status.state = "เก็บดรอป"
        collectNearbyDrops(isAlive)
        if not isAlive() then return end
    end

    local enemy, kind = pickTarget(function(v)
        return matchesName(v, q.mob) and matchesIsland(v, q.island)
    end)
    if enemy then
        Status.state = (kind == "add" and "เคลียร์มอนบริวารรอบ " or "ต่อสู้: ") .. q.mob
        fightEnemy(enemy, isAlive)
    else
        Status.state = (kind == "immune" and "บอสอมตะ รอมอนบริวาร/เปลี่ยนเฟส: " or "รอมอนเกิด: ") .. q.mob
        task.wait(0.5)
    end
    task.wait(0.2)
end

-- รายชื่อ Grimoire ที่ต้องทิ้ง = ทุก tier ที่ต่ำกว่า tier ขั้นต่ำที่เลือกเก็บ
local function getDropList()
    local keepIdx = KeepTierIndex[Settings.KeepRarity]
    local list = {}
    if not keepIdx then return list end -- "Keep All"
    for i = 1, keepIdx - 1 do
        for _, id in ipairs(RarityTiers[TierOrder[i]]) do
            table.insert(list, id)
        end
    end
    return list
end

-- ทิ้งเฉพาะตอนมีเหตุการณ์: เกมส่ง GrimoireRolled (สุ่มได้คัมภีร์) / InventoryFull (คลังเต็ม) มาให้

local dropPassRunning = false

    if ev and ev:IsA("RemoteEvent") then
        track(ev.OnClientEvent:Connect(function()
            if Settings.AutoDropGrimoire then dropRequested = true end
        end))
    end
end

local function runDropPass(isAlive)
    if dropPassRunning then return end
    dropPassRunning = true
    pcall(function()
        for _, gId in ipairs(getDropList()) do
            if not isAlive() then break end
            fire(DropGrimoireRemote, { Id = gId })
            task.wait(0.25)
        end
    end)

-- ---------- Debug: ดูข้อมูลที่เซิร์ฟเวอร์ส่งมา (เปิดดูใน F9 Console) ----------
-- ใช้เรียนรู้รูปแบบข้อมูลของ event ต่างๆ เช่น เควสต์ ร้านค้า รางวัล แล้วนำไปพัฒนาต่อได้
local DebugEvents = {
    "QuestUpdated", "QuestCompleted", "GrimoireRolled", "InventoryFull", "MaterialsEarned",
    "ChestEarned", "Notification", "MaterialShopState", "MeritAwarded", "DailyRewardState",
    "PlayTimeState", "FreeRewardState", "DungeonState", "StateChanged",
}
local debugConns = {}

local function fmtArg(v, depth)
    depth = depth or 0
    local t = typeof(v)
    if t == "table" then
        if depth >= 3 then return "{...}" end
        local parts, n = {}, 0
        for k, val in pairs(v) do
            n = n + 1
            if n > 25 then table.insert(parts, "...") break end
            table.insert(parts, tostring(k) .. "=" .. fmtArg(val, depth + 1))
        end
        return "{" .. table.concat(parts, ", ") .. "}"
    elseif t == "string" then
        return string.format("%q", v)
    elseif t == "Instance" then
        return v:GetFullName()
    end
    return tostring(v)
end

local function setDebugLog(on)
    for name, conn in pairs(debugConns) do
        pcall(function() conn:Disconnect() end)
        debugConns[name] = nil
    end
    if not on then return end
    for _, name in ipairs(DebugEvents) do
        local ev = Remotes and Remotes:FindFirstChild(name)
        if ev and ev:IsA("RemoteEvent") then
            debugConns[name] = track(ev.OnClientEvent:Connect(function(...)
                local args = table.pack(...)
                local out = {}
                for i = 1, args.n do out[i] = fmtArg(args[i]) end
                print("[LuckyHub] <- " .. name .. "(" .. table.concat(out, ", ") .. ")")
            end))
        end
    end
end

createCardToggle(ShopTab, "Debug Log (F9)", "แสดงข้อมูล event จากเซิร์ฟเวอร์ใน Console",
    Settings.DebugLog, function(state)
        Settings.DebugLog = state
        setDebugLog(state)
    end)

-- BossSummon({ Phase = "Spawn", Difficulty = ..., BossId = ... }) รูปแบบยืนยันจากล็อก remote spy
-- ค่าใช้จ่ายต่อครั้ง (Config.BossSummon): Boss Key + Yul  Easy 1 คีย์, Medium 2, Hard 3, Extreme 5
local bossSummonCount = 0

local function getSummonCfg(label)
    for _, b in ipairs(SummonBosses) do
        if b.label == label then return b end
    end
    return nil
end

-- ตรวจว่ามีบอสตัวนี้อยู่ใน Enemies แล้วหรือยัง (ไม่ต้องรอให้ตัวโหลดเข้ามาใกล้ กันเรียกซ้ำจนเสียกุญแจ)
local function findSummonedBoss(cfg)
    if not EnemiesFolder then return nil end
    for _, v in ipairs(EnemiesFolder:GetChildren()) do
        if v:IsA("Model") and (v:GetAttribute("EnemyId") == cfg.enemyId or v.Name == cfg.enemyId) then
            return v
        end
    end
    return nil
end

local function bossStep(isAlive)
    local cfg = getSummonCfg(Settings.SummonBoss)
    if not cfg then task.wait(1) return end

    if Settings.AutoCollect then
        Status.state = "เก็บดรอป"
        collectNearbyDrops(isAlive)
        if not isAlive() then return end
    end

    -- มีบอสอยู่แล้ว -> สู้ (ไม่เรียกซ้ำ)
    local existing = findSummonedBoss(cfg)
    if existing then
        local enemy, kind = pickTarget(function(v) return v == existing end)
        if enemy then
            Status.state = (kind == "add" and "เคลียร์มอนบริวารรอบ " or "ต่อสู้บอส: ") .. cfg.label
            fightEnemy(enemy, isAlive)
        else
            Status.state = (kind == "immune" and "บอสอมตะ รอมอนบริวาร/เปลี่ยนเฟส: " or "รอบอสโหลดเข้ามาใกล้: ") .. cfg.label
            task.wait(0.5)
        end
        task.wait(0.2)
        return
    end

    -- ไม่มีบอส -> เรียกใหม่ (ไม่เกินจำนวนสูงสุด)
    if bossSummonCount >= Settings.MaxSummons then
        notify("Lucky Hub", "เรียกบอสครบ " .. bossSummonCount .. " ครั้งแล้ว หยุดอัตโนมัติ", 6)
        bossToggle.Set(false)
        return
    end
    bossSummonCount = bossSummonCount + 1
    Status.state = "เรียกบอส " .. cfg.label .. " (" .. Settings.SummonDifficulty .. ") ครั้งที่ " .. bossSummonCount
    fire(BossSummonRemote, { Phase = "Spawn", Difficulty = Settings.SummonDifficulty, BossId = cfg.id })

    local t0 = os.clock()
    while isAlive() and not findSummonedBoss(cfg) and os.clock() - t0 < 20 do
        task.wait(0.5)
    end
    if isAlive() and not findSummonedBoss(cfg) then
        -- เรียกแล้วไม่ปรากฏ: กุญแจ/Yul อาจไม่พอ หรืออาจต้องอยู่จุดที่เรียกได้ หยุดทันทีเพื่อไม่ให้เสียของเปล่า
        notify("Lucky Hub", "เรียกบอสแล้วแต่ไม่ปรากฏใน 20 วินาที (กุญแจ/Yul อาจไม่พอ) หยุดอัตโนมัติ", 8)
        bossToggle.Set(false)
    end
end

-- --- TAB: RACE (สุ่มเผ่าไปเรื่อยๆ จนได้เผ่าและระดับที่เลือก) ---
-- RerollPlayerRace() ไม่มี argument (ยืนยันจากล็อก) แต่ละครั้งใช้ไอเทม K:RaceReroll หนึ่งชิ้น
-- อ่านเผ่าจาก attribute WK_PlayerRace และระดับ (Stage 1-3) จาก WK_PlayerRaceStage ของผู้เล่น
do
    local RerollState = { count = 0, noChange = 0, running = false }

    local function raceNow() return tostring(player:GetAttribute("WK_PlayerRace")) end
    local function stageNow() return tonumber(player:GetAttribute("WK_PlayerRaceStage")) or 1 end

    local function hasAnyTarget()
        for _, race in ipairs(RaceList) do
            if (Settings.RaceTargets[race] or 0) > 0 then return true end
        end
        return false
    end

    -- โอกาสได้เป้าหมายอย่างน้อยหนึ่งอย่างต่อการสุ่ม 1 ครั้ง (ยังไม่รวม Pity)
    local function targetChance()
        local p = 0
        for _, race in ipairs(RaceList) do
            local level = Settings.RaceTargets[race] or 0
            if level > 0 then p = p + RaceData[race].odds / 100 * StageAtLeast[level] end
        end
        return p
    end

    local StatusLabel2 = createInfoCard(RaceTab, 74)
    task.spawn(function()
        while Runtime.alive do
            local race, stage = raceNow(), stageNow()
            local data = RaceData[race]
            local tierName = data and data.tiers[stage] or "?"

            local goals = {}
            for _, r in ipairs(RaceList) do
                local level = Settings.RaceTargets[r] or 0
                if level > 0 then table.insert(goals, r .. " ★" .. level .. "+") end
            end
            local goalText = "ยังไม่ได้เลือก"
            if #goals > 0 then
                local p = targetChance()
                goalText = string.format("%s  (โอกาส ≈ %.1f%%/ครั้ง เฉลี่ย ~%d ครั้ง)",
                    table.concat(goals, ", "), p * 100, math.ceil(1 / p))
            end
            StatusLabel2.Text = string.format(
                "เผ่าตอนนี้: %s ★%d %s\nเป้าหมาย: %s\nสถานะ: %s",
                race, stage, tierName, goalText,
                RerollState.running and string.format("กำลังสุ่ม (ครั้งที่ %d / %d)", RerollState.count, Settings.MaxRerolls) or "หยุดอยู่"
            )
            task.wait(1)
        end
    end)

    local rerollToggle
    rerollToggle = createCardToggle(RaceTab, "เริ่มสุ่มเผ่า (สุ่มไปเรื่อยๆ จนได้เป้าหมาย)",
        "ใช้ไอเทม RaceReroll ทุกครั้ง ย้อนกลับไม่ได้! เลือกเป้าหมายด้านล่างก่อน",
        false, function(state)
            Settings.AutoRerollRace = state
            RerollState.running = state
            if not state then stopLoop("AutoRerollRace") return end

            if not hasAnyTarget() then
                notify("Lucky Hub", "ยังไม่ได้เลือกเผ่าที่ต้องการ (กด ★1/★2/★3 ที่การ์ดเผ่าด้านล่าง)", 6)
                task.defer(function() rerollToggle.Set(false) end)
                return
            end

            RerollState.count = 0
            RerollState.noChange = 0
            startLoop("AutoRerollRace", function() return Settings.AutoRerollRace end, function(isAlive)
                if not hasAnyTarget() then
                    notify("Lucky Hub", "ไม่มีเป้าหมายเหลือแล้ว หยุดสุ่ม", 6)
                    rerollToggle.Set(false)
                    return
                end
                local race, stage = raceNow(), stageNow()
                local need = Settings.RaceTargets[race] or 0
                if need > 0 and stage >= need then
                    notify("Lucky Hub", "ได้แล้ว! " .. race .. " ★" .. stage .. " (สุ่มไป " .. RerollState.count .. " ครั้ง)", 8)
                    rerollToggle.Set(false)
                    return
                end
                if RerollState.count >= Settings.MaxRerolls then
                    notify("Lucky Hub", "สุ่มครบ " .. RerollState.count .. " ครั้งแล้ว หยุดสุ่ม (ตอนนี้: " .. race .. " ★" .. stage .. ")", 8)
                    rerollToggle.Set(false)
                    return
                end
                if RerollState.noChange >= 15 then
                    -- ผลเหมือนเดิมติดกัน 15 ครั้ง (โอกาสเกิดตามธรรมชาติต่ำมาก) แปลว่าไอเทมน่าจะหมด
                    notify("Lucky Hub", "ผลไม่เปลี่ยนติดกัน 15 ครั้ง ไอเทม RaceReroll น่าจะหมด หยุดสุ่ม", 8)
                    rerollToggle.Set(false)
                    return
                end

                RerollState.count = RerollState.count + 1
                fire(RerollRaceRemote)
                -- รอให้เผ่าหรือระดับเปลี่ยน (ถ้าได้ผลเดิมหรือของหมด จะรอครบ 4 วินาทีแล้วไปรอบถัดไป)
                local t0 = os.clock()
                while isAlive() and raceNow() == race and stageNow() == stage and os.clock() - t0 < 4 do
                    task.wait(0.2)
                end
                if raceNow() == race and stageNow() == stage then
                    RerollState.noChange = RerollState.noChange + 1
                else
                    RerollState.noChange = 0
                end
                task.wait(1)
            end)
        end)

    for _, race in ipairs(RaceList) do
        createRaceCard(RaceTab, race)
    end
    createSliderCard(RaceTab, "Safety Limit (จำกัดจำนวนครั้ง กันวนไม่จบ)", 10, 500, Settings.MaxRerolls, function(val) Settings.MaxRerolls = val end)
    local RaceNote = createInfoCard(RaceTab, 44)
    RaceNote.Text = "Pity (ตีความจาก Config): Devil ครบ 40 ครั้ง, Angel ครบ 75 ครั้ง การันตีเผ่านั้น\nระดับ Stage สุ่มแยก: ★1 70% | ★2 25% | ★3 5%"
end
